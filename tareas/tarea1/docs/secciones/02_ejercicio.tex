    \subsection{Definición formal de ``Solución de una EDO''}

    Dada una EDO de la forma:
    \[ x^{(k)} = \vec{F}(t, x(t), x'(t), \dots, x^{(k-1)}(t)) \]

    Una solución de la EDO es una función $\gamma : I \to \mathbb{R}^d$ la cual es de clase $C^k$ y satisface las siguientes condiciones:

    \begin{itemize}[leftmargin=*]
        \item \textbf{i)} $I$ es un intervalo abierto.
        \item \textbf{ii)} Sea el vector $\vec{v}(t) = \left(t, \gamma(t), \frac{d}{dt}\gamma(t), \dots, \frac{d^{k-1}}{dt^{k-1}}\gamma(t)\right)$, entonces $\vec{v}(t) \subset U, \forall t \in I$.
        \item \textbf{iii)} $\frac{d^k}{dt^k}\gamma(t) = \vec{F}(\vec{v}(t))$.
    \end{itemize}
    
    Muestra que dada una EDO la función o funciones propuestas son soluciones de la EDO correspondiente.
    \item \textbf{Oscilador Armónico Amortiguado}
    \[ \frac{d^{2}x}{dt^{2}}+2\zeta\omega_{0}\frac{dx}{dt}+\omega_{0}^{2}x=0 \]
    Para interpretaciones físicas lo común es $\zeta\ge0$, $\omega_{0}>0$. Tomando $\gamma=\zeta\omega_{0}$:
    \[ \frac{d^{2}x}{dt^{2}}+2\gamma\frac{dx}{dt}+\omega_{0}^{2}x=0 \]
    Los parámetros se interpretan como:
    \begin{itemize}[leftmargin=*]
        \item $\omega_0$ (Frecuencia angular natural) Frecuencia a la que oscila el sistema en ausencia de amortiguamiento. Mide la tasa de disipación de energía en el sistema.
        \item $\gamma$ (Coeficiente de amortiguamiento o factor de amortiguamiento fraccional) Mide qué tan rápido se disipa la energía del sistema.
        \item $\zeta$ (Factor de amortiguamiento -adimensional-) Mide qué tan amortiguado está el sistema.
    \end{itemize}
    Funciones:
    \begin{enumerate}[label=(\alph*), leftmargin=*]
        \item Sin amortiguamiento $\gamma=0$, $\zeta=0$.
        \begin{equation*}
            x(t)=A \cos(\omega_{0}t)+B \sin(\omega_{0}t) \tag{1}
        \end{equation*}

        \subsection{Solución}
        \begin{itemize}[leftmargin=*]
            \item \textbf{Parámetros del caso:} $\gamma = 0$ y $\zeta = 0$.
            
            \item \textbf{Ecuación Diferencial:}
            \[ \frac{d^2x}{dt^2} + 2(0)\frac{dx}{dt} + \omega_0^2 x = 0 \implies \frac{d^2x}{dt^2} + \omega_0^2 x = 0 \]
            
            \item \textbf{Expresión en forma normal ($k = 2$):}
            \[ \frac{d^2x}{dt^2} = -\omega_0^2 x \]
            
            \item \textbf{Elementos formales:}
            \begin{itemize}[leftmargin=*]
                \item \textbf{Orden de la EDO ($k$):} $k = 2$ (segundo orden).
                \item \textbf{Dimensión del espacio ($d$):} $d = 1$ (sistema escalar, $\gamma: I \to \mathbb{R}^1$).
                \item \textbf{Función $\vec{F}$ y dominio $U$:} La función del lado derecho es $\vec{F}(t, x, x') = -\omega_0^2 x$, definida sobre el conjunto abierto $U = \mathbb{R} \times \mathbb{R} \times \mathbb{R} = \mathbb{R}^3$.
            \end{itemize}
            
            \item \textbf{Función propuesta:} $\gamma(t) = A \cos(\omega_0 t) + B \sin(\omega_0 t)$, donde $A, B \in \mathbb{R}$ son constantes arbitrarias y $\omega_0 > 0$.
        \end{itemize}

        \subsubsection*{Verificación de las Condiciones }

        \begin{itemize}[leftmargin=*]
            \item \textbf{Regularidad $C^k$ ($C^2$):}\\
            Calculamos las dos primeras derivadas de la función propuesta $\gamma(t)$:
            \begin{enumerate}[label=\arabic*., leftmargin=*]
                \item \textbf{Primera derivada:}
                \[ \frac{d}{dt}\gamma(t) = -A \omega_0 \sin(\omega_0 t) + B \omega_0 \cos(\omega_0 t) \]
                \item \textbf{Segunda derivada:}
                \[ \frac{d^2}{dt^2}\gamma(t) = -A \omega_0^2 \cos(\omega_0 t) - B \omega_0^2 \sin(\omega_0 t) \]
            \end{enumerate}
            Factorizando $-\omega_0^2$:
            \[ \frac{d^2}{dt^2}\gamma(t) = -\omega_0^2 [ A \cos(\omega_0 t) + B \sin(\omega_0 t) ] \]
            Como las funciones trigonométricas $\cos(\omega_0 t)$ y $\sin(\omega_0 t)$ son infinitamente diferenciables ($C^\infty$) en todo $\mathbb{R}$, la función $\gamma(t)$ es de clase $C^2(I)$.

            \item \textbf{Condición i) $I$ es un intervalo abierto:}\\
            Elegimos el intervalo abierto $I = (-\infty, \infty) = \mathbb{R}$.
            
            \item \textbf{Condición ii) Pertenece al dominio $U$ ($\vec{v}(t) \in U, \forall t \in I$):}\\
            Para $k = 2$, el vector de estado $\vec{v}(t)$ está definido por:
            \[ \vec{v}(t) = \left(t, \gamma(t), \frac{d}{dt}\gamma(t)\right) = \left(t, A \cos(\omega_0 t) + B \sin(\omega_0 t), -A \omega_0 \sin(\omega_0 t) + B \omega_0 \cos(\omega_0 t)\right) \]
            Dado que $U = \mathbb{R}^3$, para cualquier valor $t \in I = \mathbb{R}$, el vector $\vec{v}(t) \in \mathbb{R}^3 = U$. Por lo tanto, $\vec{v}(t) \in U, \forall t \in I$.
            
            \item \textbf{Condición iii) Satisfacción de la ecuación ($\frac{d^2}{dt^2}\gamma(t) = \vec{F}(\vec{v}(t))$):}
            \begin{itemize}[leftmargin=*]
                \item \textbf{Lado Izquierdo (derivada de orden $k=2$):}
                \[ \frac{d^2}{dt^2}\gamma(t) = -\omega_0^2 [ A \cos(\omega_0 t) + B \sin(\omega_0 t) ] \]
                \item \textbf{Lado Derecho ($\vec{F}(\vec{v}(t))$):}
                \[ \vec{F}(\vec{v}(t)) = \vec{F}\left(t, \gamma(t), \gamma'(t)\right) = -\omega_0^2 \gamma(t) = -\omega_0^2 [ A \cos(\omega_0 t) + B \sin(\omega_0 t) ] \]
                \item \textbf{Identidad:}
                \[ -\omega_0^2 [ A \cos(\omega_0 t) + B \sin(\omega_0 t) ] = -\omega_0^2 [ A \cos(\omega_0 t) + B \sin(\omega_0 t) ] \]
            \end{itemize}
        \end{itemize}

        \vspace{0.3cm}
        \noindent La función $\gamma(t) = A \cos(\omega_0 t) + B \sin(\omega_0 t)$ satisface rigurosamente las condiciones (i, ii y iii) de la definición formal, demostrando que es una solución explícita de clase $C^2$ en el intervalo abierto $I = \mathbb{R}$ para la EDO del oscilador armónico sin amortiguamiento.

        \item Caso sub-amortiguado, $\gamma<\omega_{0}$, $0<\zeta<1$.
        \begin{equation*}
            \omega_{d}=\sqrt{\omega_{0}^{2}-\gamma^{2}} \tag{2}
        \end{equation*}
        \begin{equation*}
            x(t)=e^{-\gamma t}[A \cos(\omega_{d}t)+B \sin(\omega_{d}t)] \tag{3}
        \end{equation*}

        \subsection{Solución}
        \begin{itemize}[leftmargin=*]
            \item \textbf{Parámetros del caso:} $\gamma < \omega_0$ y $0 < \zeta < 1$.
            
            \item \textbf{Ecuación Diferencial:}
            \[ \frac{d^2x}{dt^2} + 2\gamma\frac{dx}{dt} + \omega_0^2 x = 0 \]
            
            \item \textbf{Expresión en forma normal ($k = 2$):}
            \[ \frac{d^2x}{dt^2} = -2\gamma\frac{dx}{dt} - \omega_0^2 x \]
            
            \item \textbf{Elementos formales:}
            \begin{itemize}[leftmargin=*]
                \item \textbf{Orden de la EDO ($k$):} $k = 2$ (segundo orden).
                \item \textbf{Dimensión del espacio ($d$):} $d = 1$ (sistema escalar, $\gamma: I \to \mathbb{R}^1$).
                \item \textbf{Función $\vec{F}$ y dominio $U$:} La función del lado derecho es $\vec{F}(t, x, x') = -2\gamma x' - \omega_0^2 x$, definida sobre el conjunto abierto $U = \mathbb{R}^3$.
                \item \textbf{Función propuesta:} $\gamma(t) = e^{-\gamma t}[A \cos(\omega_d t) + B \sin(\omega_d t)]$, donde la frecuencia angular amortiguada es $\omega_d = \sqrt{\omega_0^2 - \gamma^2}$.
            \end{itemize}
        \end{itemize}

        \subsubsection*{Verificación de las Condiciones}

        \begin{itemize}[leftmargin=*]
            \item \textbf{Regularidad $C^k$ ($C^2$):}\\
            Para simplificar la notación de las derivadas, definimos la parte oscilatoria como $u(t) = A \cos(\omega_d t) + B \sin(\omega_d t)$. Sabemos por el inciso anterior que su segunda derivada es $u''(t) = -\omega_d^2 u(t)$.\\
            Nuestra función propuesta es $\gamma(t) = e^{-\gamma t}u(t)$. Calculamos sus derivadas:
            \begin{enumerate}[label=\arabic*., leftmargin=*]
                \item \textbf{Primera derivada (Regla del producto):}
                \[ \frac{d}{dt}\gamma(t) = e^{-\gamma t}[u'(t) - \gamma u(t)] \]
                \item \textbf{Segunda derivada (Regla del producto):}
                \[ \frac{d^2}{dt^2}\gamma(t) = -\gamma e^{-\gamma t}[u'(t) - \gamma u(t)] + e^{-\gamma t}[u''(t) - \gamma u'(t)] \]
            \end{enumerate}
            Agrupando términos y factorizando $e^{-\gamma t}$:
            \[ \frac{d^2}{dt^2}\gamma(t) = e^{-\gamma t} [ \gamma^2 u(t) - 2\gamma u'(t) + u''(t) ] \]
            Sustituyendo $u''(t) = -\omega_d^2 u(t)$:
            \[ \frac{d^2}{dt^2}\gamma(t) = e^{-\gamma t} [ (\gamma^2 - \omega_d^2)u(t) - 2\gamma u'(t) ] \]
            Como $\omega_d^2 = \omega_0^2 - \gamma^2$, sabemos que $\gamma^2 - \omega_d^2 = 2\gamma^2 - \omega_0^2$. Sustituyendo esto último:
            \[ \frac{d^2}{dt^2}\gamma(t) = e^{-\gamma t} [ (2\gamma^2 - \omega_0^2)u(t) - 2\gamma u'(t) ] \]
            Dado que las funciones exponenciales y trigonométricas son de clase $C^\infty$ en todo $\mathbb{R}$, la función $\gamma(t)$ es rigurosamente de clase $C^2(I)$.

            \item \textbf{Condición i) $I$ es un intervalo abierto:}\\
            Definimos el intervalo abierto como $I = (-\infty, \infty) = \mathbb{R}$.

            \item \textbf{Condición ii) Pertenece al dominio $U$ ($\vec{v}(t) \in U, \forall t \in I$):}\\
            El vector de estado para $k=2$ es:
            \[ \vec{v}(t) = \left(t, \gamma(t), \frac{d}{dt}\gamma(t)\right) \]
            Puesto que $U = \mathbb{R}^3$, el vector $\vec{v}(t)$ pertenece a $U$ para cualquier instante $t \in \mathbb{R}$.

            \item \textbf{Condición iii) Satisfacción de la ecuación ($\frac{d^2}{dt^2}\gamma(t) = \vec{F}(\vec{v}(t))$):}
            \begin{itemize}[leftmargin=*]
                \item \textbf{Lado Izquierdo (derivada de orden $k=2$):}
                \[ \frac{d^2}{dt^2}\gamma(t) = e^{-\gamma t} [ (2\gamma^2 - \omega_0^2)u(t) - 2\gamma u'(t) ] \]
                \item \textbf{Lado Derecho ($\vec{F}(\vec{v}(t))$):}\\
                Evaluamos la función $\vec{F}$ en nuestro vector de estado:
                \[ \vec{F}(\vec{v}(t)) = -2\gamma \gamma'(t) - \omega_0^2 \gamma(t) \]
                Sustituimos $\gamma(t)$ y $\gamma'(t)$:
                \[ \vec{F}(\vec{v}(t)) = -2\gamma \left( e^{-\gamma t}[u'(t) - \gamma u(t)] \right) - \omega_0^2 \left( e^{-\gamma t}u(t) \right) \]
                Factorizamos $e^{-\gamma t}$:
                \[ \vec{F}(\vec{v}(t)) = e^{-\gamma t} [ -2\gamma u'(t) + 2\gamma^2 u(t) - \omega_0^2 u(t) ] \]
                Agrupando los términos que multiplican a $u(t)$:
                \[ \vec{F}(\vec{v}(t)) = e^{-\gamma t} [ (2\gamma^2 - \omega_0^2)u(t) - 2\gamma u'(t) ] \]
                \item \textbf{Identidad:}
                \[ e^{-\gamma t} [ (2\gamma^2 - \omega_0^2)u(t) - 2\gamma u'(t) ] = e^{-\gamma t} [ (2\gamma^2 - \omega_0^2)u(t) - 2\gamma u'(t) ] \]
            \end{itemize}
        \end{itemize}

        \vspace{0.3cm}
        \noindent La igualdad se cumple de manera exacta para todo $t \in \mathbb{R}$. La función $\gamma(t) = e^{-\gamma t}[A \cos(\omega_d t) + B \sin(\omega_d t)]$ satisface las tres condiciones de la definición formal, comprobando que es solución explícita para la EDO del oscilador armónico sub-amortiguado.
        
        \item Amortiguamiento crítico, $\gamma=\omega_{0}$, $0<\zeta<1$.
        \begin{equation*}
            x(t)=(A+Bt)e^{-\gamma t} \tag{4}
        \end{equation*}

        \subsection{Solución}
        \begin{itemize}[leftmargin=*]
            \item \textbf{Parámetros del caso:} $\gamma = \omega_0$.
            
            \item \textbf{Ecuación Diferencial:}\\
            Como $\gamma = \omega_0$, sustituimos $\omega_0^2$ por $\gamma^2$ en la ecuación original para simplificar la comprobación:
            \[ \frac{d^2x}{dt^2} + 2\gamma\frac{dx}{dt} + \gamma^2 x = 0 \]
            
            \item \textbf{Expresión en forma normal ($k = 2$):}
            \[ \frac{d^2x}{dt^2} = -2\gamma\frac{dx}{dt} - \gamma^2 x \]
            
            \item \textbf{Elementos formales:}
            \begin{itemize}[leftmargin=*]
                \item \textbf{Orden de la EDO ($k$):} $k = 2$ (segundo orden).
                \item \textbf{Dimensión del espacio ($d$):} $d = 1$ (sistema escalar, $x: I \to \mathbb{R}^1$).
                \item \textbf{Función $\vec{F}$ y dominio $U$:} La función es $\vec{F}(t, x, x') = -2\gamma x' - \gamma^2 x$, la cual está definida de manera continua sobre el conjunto abierto $U = \mathbb{R}^3$.
                \item \textbf{Función propuesta:} $x(t) = (A + Bt)e^{-\gamma t}$, donde $A, B \in \mathbb{R}$ son las constantes determinadas por las condiciones iniciales.
            \end{itemize}
        \end{itemize}

        \vspace{0.3cm}
        \noindent\textbf{Verificación de las Condiciones}

        \begin{itemize}[leftmargin=*]
            \item \textbf{Regularidad $C^k$ ($C^2$):}\\
            Calculamos las dos primeras derivadas de la función propuesta $x(t) = (A + Bt)e^{-\gamma t}$:
            \begin{enumerate}[label=\arabic*., leftmargin=*]
                \item \textbf{Primera derivada (Regla del producto):}
                \[ \frac{d}{dt}x(t) = B e^{-\gamma t} - \gamma(A + Bt)e^{-\gamma t} \]
                Factorizamos $e^{-\gamma t}$:
                \[ \frac{d}{dt}x(t) = e^{-\gamma t} [ B - \gamma A - \gamma B t ] \]
                
                \item \textbf{Segunda derivada (Regla del producto aplicada al paso anterior):}
                \[ \frac{d^2}{dt^2}x(t) = -\gamma B e^{-\gamma t} - \gamma e^{-\gamma t} [ B - \gamma A - \gamma B t ] \]
                Agrupando y factorizando $e^{-\gamma t}$:
                \[ \frac{d^2}{dt^2}x(t) = e^{-\gamma t} [ -2\gamma B + \gamma^2 A + \gamma^2 B t ] \]
            \end{enumerate}
            Dado que los polinomios algebraicos y las funciones exponenciales son suaves e infinitamente diferenciables ($C^\infty$) en todo $\mathbb{R}$, se cumple rigurosamente que $x(t)$ es de clase $C^2(I)$.

            \item \textbf{Condición i) $I$ es un intervalo abierto:}\\
            Definimos nuestro intervalo temporal como toda la recta real, $I = (-\infty, \infty) = \mathbb{R}$.

            \item \textbf{Condición ii) Pertenece al dominio $U$ ($\vec{v}(t) \in U, \forall t \in I$):}\\
            Construimos el vector de estado para $k=2$:
            \[ \vec{v}(t) = \left(t, x(t), \frac{d}{dt}x(t)\right) \]
            Dado que $t$, $x(t)$ y su derivada arrojan valores reales bien definidos para cualquier $t$, el vector $\vec{v}(t)$ pertenece inequívocamente al dominio $U = \mathbb{R}^3$ para todo instante $t \in I$.

            \item \textbf{Condición iii) Satisfacción de la ecuación ($\frac{d^2}{dt^2}x(t) = \vec{F}(\vec{v}(t))$):}
            \begin{itemize}[leftmargin=*]
                \item \textbf{Lado Izquierdo (derivada de orden $k=2$ previamente calculada):}
                \[ \frac{d^2}{dt^2}x(t) = e^{-\gamma t} [ -2\gamma B + \gamma^2 A + \gamma^2 B t ] \]
                
                \item \textbf{Lado Derecho ($\vec{F}(\vec{v}(t))$):}\\
                Evaluamos la función normalizada $\vec{F}$ sustituyendo $x(t)$ y $x'(t)$:
                \[ \vec{F}(\vec{v}(t)) = -2\gamma \left( e^{-\gamma t} [ B - \gamma A - \gamma B t ] \right) - \gamma^2 \left( (A + Bt)e^{-\gamma t} \right) \]
                Factorizamos el término $e^{-\gamma t}$ común a ambas expresiones:
                \[ \vec{F}(\vec{v}(t)) = e^{-\gamma t} [ -2\gamma B + 2\gamma^2 A + 2\gamma^2 B t - \gamma^2 A - \gamma^2 B t ] \]
                Reducimos los términos algebraicos semejantes dentro del corchete ($2\gamma^2 A - \gamma^2 A$ y $2\gamma^2 B t - \gamma^2 B t$):
                \[ \vec{F}(\vec{v}(t)) = e^{-\gamma t} [ -2\gamma B + \gamma^2 A + \gamma^2 B t ] \]
                
                \item \textbf{Identidad:}
                \[ e^{-\gamma t} [ -2\gamma B + \gamma^2 A + \gamma^2 B t ] = e^{-\gamma t} [ -2\gamma B + \gamma^2 A + \gamma^2 B t ] \]
            \end{itemize}
        \end{itemize}

        \vspace{0.3cm}
        \noindent La igualdad analítica se cumple de forma exacta, lo que comprueba formalmente que $x(t) = (A + Bt)e^{-\gamma t}$ es una solución de la EDO para el oscilador armónico críticamente amortiguado.
        \item Caso sobre-amortiguado, $\gamma>\omega_{0}$, $\zeta>1$.
        \[ x(t)=Ae^{-\alpha_{1}t}+Be^{-\alpha_{2}t} \]
        \begin{equation*}
            \alpha_{1}=\gamma+\sqrt{\gamma^{2}-\omega_{0}^{2}} \quad \text{y} \quad \alpha_{2}=\gamma-\sqrt{\gamma^{2}-\omega_{0}^{2}}. \tag{5}
        \end{equation*}

        \subsection{Solución}
        \begin{itemize}[leftmargin=*]
            \item \textbf{Parámetros del caso:} $\gamma > \omega_0$ y $\zeta > 1$.
            
            \item \textbf{Ecuación Diferencial:}
            \[ \frac{d^2x}{dt^2} + 2\gamma\frac{dx}{dt} + \omega_0^2 x = 0 \]
            
            \item \textbf{Expresión en forma normal ($k = 2$):}
            \[ \frac{d^2x}{dt^2} = -2\gamma\frac{dx}{dt} - \omega_0^2 x \]
            
            \item \textbf{Elementos formales:}
            \begin{itemize}[leftmargin=*]
                \item \textbf{Orden de la EDO ($k$):} $k = 2$ (segundo orden).
                \item \textbf{Dimensión del espacio ($d$):} $d = 1$ (sistema escalar, $x: I \to \mathbb{R}^1$).
                \item \textbf{Función $\vec{F}$ y dominio $U$:} La función $\vec{F}(t, x, x') = -2\gamma x' - \omega_0^2 x$ está bien definida y es continua en el dominio abierto $U = \mathbb{R}^3$.
                \item \textbf{Función propuesta:} $x(t) = A e^{-\alpha_1 t} + B e^{-\alpha_2 t}$, donde las constantes asociadas a las raíces del polinomio característico son $\alpha_1 = \gamma + \sqrt{\gamma^2 - \omega_0^2}$ y $\alpha_2 = \gamma - \sqrt{\gamma^2 - \omega_0^2}$.
            \end{itemize}
        \end{itemize}

        \subsubsection*{Verificación Paso a Paso de las Condiciones Formales}

        \begin{itemize}[leftmargin=*]
            \item \textbf{Regularidad $C^k$ ($C^2$):}\\
            Calculamos las derivadas sucesivas de la función propuesta $x(t) = A e^{-\alpha_1 t} + B e^{-\alpha_2 t}$:
            \begin{enumerate}[label=\arabic*., leftmargin=*]
                \item \textbf{Primera derivada:}
                \[ \frac{d}{dt}x(t) = -A\alpha_1 e^{-\alpha_1 t} - B\alpha_2 e^{-\alpha_2 t} \]
                \item \textbf{Segunda derivada:}
                \[ \frac{d^2}{dt^2}x(t) = A\alpha_1^2 e^{-\alpha_1 t} + B\alpha_2^2 e^{-\alpha_2 t} \]
            \end{enumerate}
            Al estar conformada por la suma de dos funciones exponenciales reales, las cuales son infinitamente diferenciables ($C^\infty$) en cualquier instante del tiempo, la función $x(t)$ pertenece invariablemente a la clase $C^2(I)$.

            \item \textbf{Condición i) $I$ es un intervalo abierto:}\\
            Tomamos como intervalo de existencia toda la recta real, es decir, $I = (-\infty, \infty) = \mathbb{R}$.

            \item \textbf{Condición ii) Pertenece al dominio $U$ ($\vec{v}(t) \in U, \forall t \in I$):}\\
            Construimos el vector de estado para $k=2$:
            \[ \vec{v}(t) = \left(t, x(t), \frac{d}{dt}x(t)\right) \]
            Dado que el tiempo, la posición $x(t)$ y la velocidad $x'(t)$ son números reales escalares, el vector ordenado $\vec{v}(t)$ pertenece a $U = \mathbb{R}^3$ para todo valor de $t \in \mathbb{R}$.

            \item \textbf{Condición iii) Satisfacción de la ecuación ($\frac{d^2}{dt^2}x(t) = \vec{F}(\vec{v}(t))$):}
            \begin{itemize}[leftmargin=*]
                \item \textbf{Lado Izquierdo (derivada de orden $k=2$ calculada previamente):}
                \[ \frac{d^2}{dt^2}x(t) = A\alpha_1^2 e^{-\alpha_1 t} + B\alpha_2^2 e^{-\alpha_2 t} \]
                
                \item \textbf{Lado Derecho ($\vec{F}(\vec{v}(t))$):}\\
                Evaluamos la función $\vec{F}$ empleando $x(t)$ y $x'(t)$:
                \[ \vec{F}(\vec{v}(t)) = -2\gamma \left( -A\alpha_1 e^{-\alpha_1 t} - B\alpha_2 e^{-\alpha_2 t} \right) - \omega_0^2 \left( A e^{-\alpha_1 t} + B e^{-\alpha_2 t} \right) \]
                Agrupamos los términos de acuerdo a sus factores exponenciales correspondientes $A e^{-\alpha_1 t}$ y $B e^{-\alpha_2 t}$:
                \[ \vec{F}(\vec{v}(t)) = A e^{-\alpha_1 t}(2\gamma\alpha_1 - \omega_0^2) + B e^{-\alpha_2 t}(2\gamma\alpha_2 - \omega_0^2) \]
                
                \textit{(Para facilitar la demostración, notemos una propiedad de $\alpha_1$ y $\alpha_2$. Si elevamos $\alpha_1$ al cuadrado obtenemos $\alpha_1^2 = (\gamma + \sqrt{\gamma^2 - \omega_0^2})^2 = \gamma^2 + 2\gamma\sqrt{\gamma^2 - \omega_0^2} + \gamma^2 - \omega_0^2 = 2\gamma^2 + 2\gamma\sqrt{\gamma^2 - \omega_0^2} - \omega_0^2$. Sabiendo que $2\gamma\alpha_1 = 2\gamma^2 + 2\gamma\sqrt{\gamma^2 - \omega_0^2}$, podemos deducir que \textbf{$\alpha_1^2 = 2\gamma\alpha_1 - \omega_0^2$}. El mismo desarrollo exacto aplica para $\alpha_2$.)}

                Sustituyendo la identidad algebraica descrita anteriormente ($\alpha^2 = 2\gamma\alpha - \omega_0^2$), la expresión se simplifica a:
                \[ \vec{F}(\vec{v}(t)) = A e^{-\alpha_1 t}(\alpha_1^2) + B e^{-\alpha_2 t}(\alpha_2^2) = A\alpha_1^2 e^{-\alpha_1 t} + B\alpha_2^2 e^{-\alpha_2 t} \]
                
                \item \textbf{Identidad:}
                \[ A\alpha_1^2 e^{-\alpha_1 t} + B\alpha_2^2 e^{-\alpha_2 t} = A\alpha_1^2 e^{-\alpha_1 t} + B\alpha_2^2 e^{-\alpha_2 t} \]
            \end{itemize}
        \end{itemize}

        \vspace{0.3cm}
        \noindent La identidad se satisface exactamente. De esta manera, demostramos que $x(t) = A e^{-\alpha_1 t} + B e^{-\alpha_2 t}$ es, en efecto, una solución de clase $C^2$ a la EDO del sistema en el caso de oscilación sobre-amortiguada.
    \end{enumerate}

    \item \textbf{Dinámica Poblacional con Factor (capacidad) de Carga. Ecuación Logística o de Verhulst}
    
    La tasa de cambio de una población $P$ con respecto al tiempo $t$, dado que al tiempo $t_{0}$ cuenta con $P_{0}$ integrantes, esta dada por la EDO:
    \[ \frac{dP}{dt}=rP\left(1-\frac{P}{K}\right) \]
    \[ P(t_{0})=P_{0} \quad \text{(Condiciones iniciales)} \]
    donde $K>0$ es ``carrying capacity'' y $r>0$ ``growth rate'', ambas constantes.
    \begin{enumerate}[label=(\alph*), leftmargin=*]
        \item Función:
        \[ P(t)=\frac{KP_{0}}{P_{0}+(K-P_{0})e^{r(t_{0}-t)}} \]

        \subsection*{Solución}

        \begin{itemize}[leftmargin=*]
            \item \textbf{Parámetros del caso:} $K > 0$ (capacidad de carga), $r > 0$ (tasa de crecimiento) y $P_0$ (población inicial en $t_0$).
            
            \item \textbf{Ecuación Diferencial:}
            \[ \frac{dP}{dt} = rP\left(1 - \frac{P}{K}\right) \]
            
            \item \textbf{Expresión en forma normal ($k = 1$):}
            \[ \frac{dP}{dt} = rP - \frac{r}{K}P^2 \]
            
            \item \textbf{Elementos formales:}
            \begin{itemize}[leftmargin=*]
                \item \textbf{Orden de la EDO ($k$):} $k = 1$ (primer orden).
                \item \textbf{Dimensión del espacio ($d$):} $d = 1$ (sistema escalar, $\gamma: I \to \mathbb{R}^1$).
                \item \textbf{Función $\vec{F}$ y dominio $U$:} La función del lado derecho es $\vec{F}(t, P) = rP(1 - P/K)$, definida sobre el conjunto abierto $U = \mathbb{R}^2$.
            \end{itemize}
            
            \item \textbf{Función propuesta:} $\gamma(t) = \frac{K P_0}{P_0 + (K - P_0)e^{r(t_0 - t)}}$
        \end{itemize}

        \subsubsection*{Verificación de las Condiciones}

        \begin{itemize}[leftmargin=*]
            \item \textbf{Regularidad $C^k$ ($C^1$):}\\
            Para simplificar la derivación, definimos el denominador como $D(t) = P_0 + (K - P_0)e^{r(t_0 - t)}$. Su derivada respecto al tiempo es $D'(t) = -r(K - P_0)e^{r(t_0 - t)}$.\\
            Reescribimos la función como $\gamma(t) = K P_0 [D(t)]^{-1}$ y aplicamos la regla de la cadena para obtener su primera derivada:
            \[ \frac{d}{dt}\gamma(t) = -K P_0 [D(t)]^{-2} \cdot D'(t) \]
            Sustituimos $D'(t)$:
            \[ \frac{d}{dt}\gamma(t) = \frac{r K P_0 (K - P_0)e^{r(t_0 - t)}}{[P_0 + (K - P_0)e^{r(t_0 - t)}]^2} \]
            Asumiendo poblaciones con sentido físico ($P_0 > 0$), el denominador nunca se anula. Al estar compuesta por funciones exponenciales y algebraicas suaves, $\gamma(t)$ es de clase $C^\infty$, cumpliendo con ser $C^1(I)$.

            \item \textbf{Condición i) $I$ es un intervalo abierto:}\\
            Definimos el intervalo abierto $I$ como toda la recta real $\mathbb{R}$ (o el subconjunto temporal máximo donde el denominador no sea cero).

            \item \textbf{Condición ii) Pertenece al dominio $U$ ($\vec{v}(t) \in U, \forall t \in I$):}\\
            Para $k = 1$, el vector de estado se reduce a:
            \[ \vec{v}(t) = (t, \gamma(t)) \]
            Dado que $U = \mathbb{R}^2$, el par ordenado $\vec{v}(t)$ pertenece a $U$ para cualquier instante $t \in I$.

            \item \textbf{Condición iii) Satisfacción de la ecuación ($\frac{d}{dt}\gamma(t) = \vec{F}(\vec{v}(t))$):}
            \begin{itemize}[leftmargin=*]
                \item \textbf{Lado Izquierdo (derivada calculada):}
                \[ \frac{d}{dt}\gamma(t) = \frac{r K P_0 (K - P_0)e^{r(t_0 - t)}}{[D(t)]^2} \]
                
                \item \textbf{Lado Derecho ($\vec{F}(\vec{v}(t))$):}\\
                Evaluamos la función $\vec{F}$ sustituyendo $\gamma(t)$:
                \[ \vec{F}(\vec{v}(t)) = r\gamma(t) \left( 1 - \frac{\gamma(t)}{K} \right) \]
                Calculamos el término del paréntesis por separado:
                \[ 1 - \frac{\gamma(t)}{K} = 1 - \frac{P_0}{D(t)} = \frac{D(t) - P_0}{D(t)} \]
                Sustituimos el valor de $D(t) = P_0 + (K - P_0)e^{r(t_0 - t)}$ en el numerador:
                \[ 1 - \frac{\gamma(t)}{K} = \frac{P_0 + (K - P_0)e^{r(t_0 - t)} - P_0}{D(t)} = \frac{(K - P_0)e^{r(t_0 - t)}}{D(t)} \]
                Ahora multiplicamos este resultado por $r\gamma(t)$, recordando que $r\gamma(t) = \frac{r K P_0}{D(t)}$:
                \[ \vec{F}(\vec{v}(t)) = \left( \frac{r K P_0}{D(t)} \right) \left( \frac{(K - P_0)e^{r(t_0 - t)}}{D(t)} \right) = \frac{r K P_0 (K - P_0)e^{r(t_0 - t)}}{[D(t)]^2} \]
                
                \item \textbf{Identidad:}\\
                Ambos lados de la ecuación son idénticos, garantizando que la EDO se satisface rigurosamente.
            \end{itemize}
        \end{itemize}

        \subsubsection*{Comprobación de la Condición Inicial}

        El problema también pide que se cumpla la condición inicial $P(t_0) = P_0$. Así que, evaluamos nuestra función $\gamma(t)$ en el instante $t = t_0$:

        \begin{align*}
            \gamma(t_0) &= \frac{K P_0}{P_0 + (K - P_0)e^{r(t_0 - t_0)}} \\
            \gamma(t_0) &= \frac{K P_0}{P_0 + (K - P_0)e^{0}} \\
            \gamma(t_0) &= \frac{K P_0}{P_0 + (K - P_0)(1)} = \frac{K P_0}{P_0 + K - P_0} \\
            \gamma(t_0) &= \frac{K P_0}{K} = P_0
        \end{align*}

        \vspace{0.2cm}
        \noindent Por lo tanto, hemos demostrado que la función cumple tanto con la EDO como con la condición inicial.

        \item Muestra que si $t_{0}=0$ en la función anterior, entonces esta se puede reescribir en la forma vista en clase:
        \[ P(t)=\frac{KP_{0}e^{rt}}{K+P_{0}(e^{rt}-1)} \]

        \subsection*{Solución}

        Empezamos con la función original:

        \[ P(t) = \frac{K P_0}{P_0 + (K - P_0)e^{r(t_0 - t)}} \]

        El tiempo inicial es $t_0 = 0$. Al sustituir este valor, obtenemos:

        \begin{align*}
            P(t) &= \frac{K P_0}{P_0 + (K - P_0)e^{r(0 - t)}} \\
            P(t) &= \frac{K P_0}{P_0 + (K - P_0)e^{-rt}}
        \end{align*}

        Multiplicamos tanto el numerador como el denominador por el factor $e^{rt}$:

        \[ P(t) = \frac{K P_0 \cdot e^{rt}}{[P_0 + (K - P_0)e^{-rt}] \cdot e^{rt}} \]

        Aplicamos la propiedad distributiva en el denominador:

        \[ P(t) = \frac{K P_0 e^{rt}}{P_0 e^{rt} + (K - P_0)e^{-rt}e^{rt}} \]

        Recordando por las leyes de los exponentes que $e^{-rt}e^{rt} = e^0 = 1$, el término de la derecha se simplifica:

        \[ P(t) = \frac{K P_0 e^{rt}}{P_0 e^{rt} + K - P_0} \]

        Reordenamos el denominador para agrupar los términos que contienen la constante $P_0$:

        \[ P(t) = \frac{K P_0 e^{rt}}{K + P_0 e^{rt} - P_0} \]

        Finalmente, factorizamos $P_0$ en el denominador:

        \[ P(t) = \frac{K P_0 e^{rt}}{K + P_0(e^{rt} - 1)} \]

        Por lo tanto, hemos demostrado cómo la función general evaluada en $t_0 = 0$ es equivalente a la expresión requerida.
        \item ¿Qué tipo de condiciones iniciales tiene la ecuación del inciso b?
        
        \subsection*{Solución}

        Evaluamos la función en el instante de tiempo $t = 0$.

        Iniciamos con:

        \[ P(t) = \frac{KP_0e^{rt}}{K + P_0(e^{rt} - 1)} \]

        Sustituimos $t = 0$:

        \begin{align*}
            P(0) &= \frac{KP_0e^{r(0)}}{K + P_0(e^{r(0)} - 1)} \\
            P(0) &= \frac{KP_0(1)}{K + P_0(1 - 1)} \\
            P(0) &= \frac{KP_0}{K + 0} \\
            P(0) &= P_0
        \end{align*}

        \textbf{Conclusión:} La ecuación del inciso (b) obedece a una condición inicial donde el tiempo de inicio está fijado en el origen ($t_0 = 0$) y la población en ese instante exacto es $P_0$. Es decir, asume explícitamente la condición $P(0) = P_0$.
        
        \item Cuál de las funciones es más general (da todas las soluciones).
        \subsection*{Solución}

        Al comparar ambas expresiones matemáticas, \textbf{la función del inciso (a) es la más general}.

        \begin{itemize}[leftmargin=*]
            \item \textbf{Función (a):} $P(t) = \frac{KP_0}{P_0 + (K - P_0)e^{r(t_0 - t)}}$
            \item \textbf{Función (b):} $P(t) = \frac{KP_0e^{rt}}{K + P_0(e^{rt} - 1)}$
        \end{itemize}

        La función del inciso (a) incluye el parámetro $t_0$, lo que le permite describir la dinámica poblacional a partir de \textit{cualquier} instante de tiempo inicial arbitrario. Por el contrario, la función del inciso (b) es un caso particular derivado directamente de forzar $t_0 = 0$. Por lo tanto, la primera fórmula abarca a todas las soluciones posibles sin importar la elección del tiempo de inicial.
    \end{enumerate}

    \item \textbf{EDO Separable no lineal.}
    \[ \frac{dy}{dx}=\frac{4-2x}{3y^{2}-5} \]
    \begin{enumerate}[label=(\alph*), leftmargin=*]
        \item Función:
        \[ y^{3}-5y+x^{2}-4x=c \]

        \subsection*{Solución}

        Para comprobar que la anterior función es la solución de la ecuación diferencial, aplicamos derivación implícita con respecto a la variable independiente $x$ en ambos lados de la igualdad.

        \[ \frac{d}{dx} (y^3 - 5y + x^2 - 4x) = \frac{d}{dx} (c) \]

        Aplicando la regla de la cadena para los términos que contienen a la variable dependiente $y$:

        \[ 3y^2 \frac{dy}{dx} - 5 \frac{dy}{dx} + 2x - 4 = 0 \]

        Agrupamos y factorizamos el término común $\frac{dy}{dx}$:

        \[ \frac{dy}{dx} (3y^2 - 5) + 2x - 4 = 0 \]
        \[ \frac{dy}{dx} (3y^2 - 5) = 4 - 2x \]
        \[ \frac{dy}{dx} = \frac{4 - 2x}{3y^2 - 5} \]

        La expresión coincide con la ecuación diferencial original. Por lo que, hemos demostrado que es solución.

        \item ¿Qué valores nunca se pueden tomar? ¿Son puntos del demonio o funciones?
        
        \subsection*{Solución}

        \[ \frac{dy}{dx} = \frac{4 - 2x}{3y^2 - 5} \]

        La expresión matemática se indefine (causa una división por cero) cuando el denominador se anula. Para encontrar estos valores, igualamos el denominador a cero y despejamos:

        \begin{align*}
            3y^2 - 5 &= 0 \\
            3y^2 &= 5 \\
            y^2 &= \frac{5}{3} \\
            y &= \pm\sqrt{\frac{5}{3}}
        \end{align*}

        Por lo tanto, los valores que la variable dependiente nunca puede alcanzar son \textbf{$y = \sqrt{\frac{5}{3}}$} y \textbf{$y = -\sqrt{\frac{5}{3}}$}.

        \vspace{0.3cm}
        \noindent\textbf{¿Son puntos del demonio o funciones?}\\
        Estos valores representan \textbf{funciones} (específicamente, funciones constantes), no ``puntos del demonio''.

        Un ``punto del demonio'' es una coordenada específica $(x_0, y_0)$ en el plano donde el sistema colapsa o se indefine de manera puntual. En nuestro caso, la restricción no depende de $x$; la indefinición ocurre para \textit{cualquier} valor de $x$ siempre que $y$ tome esos valores críticos. Geométricamente, esto significa que las restricciones no son puntos aislados, sino dos rectas horizontales continuas ($y = 1.29$ y $y = -1.29$ aproximadamente) que cruzan todo el plano cartesiano.
        
        \item Qué condición impone a las curvas integrales los valores determinados en el inciso anterior.
        
        Al analizar la ecuación diferencial, resulta evidente que cuando la variable dependiente toma estos valores (\textbf{$y = \sqrt{\frac{5}{3}}$} y \textbf{$y = -\sqrt{\frac{5}{3}}$}), el denominador se anula y la pendiente de la curva tiende a infinito. Esto significa que, al aproximarse a estas líneas horizontales, las curvas integrales desarrollan tangentes completamente verticales. Sin embargo, por definición matemática, la gráfica de una solución derivable no puede tener una línea tangente vertical en ninguna parte. En consecuencia, los valores que determinamos actúan como una barrera que fragmenta el plano; las soluciones quedan confinadas a regiones específicas y no pueden cruzar estas fronteras, limitando así el intervalo de definición en el que una solución particular es válida y continua.
    \end{enumerate}

    \item \textbf{Ecuación de Bernoulli}
    \[ \frac{dz}{dt}+z=z^{2} \]
    Función:
    \[ z(t)=\frac{1}{1+Ce^{t}} \]
    \subsection*{Solución}

    \begin{itemize}[leftmargin=*]
        \item \textbf{Ecuación Diferencial:}
        \[ \frac{dz}{dt} + z = z^2 \]
        
        \item \textbf{Expresión en forma normal ($k = 1$):}
        \[ \frac{dz}{dt} = z^2 - z \]
        
        \item \textbf{Elementos formales:}
        \begin{itemize}[leftmargin=*]
            \item \textbf{Orden de la EDO ($k$):} $k = 1$ (primer orden).
            \item \textbf{Dimensión del espacio ($d$):} $d = 1$ (sistema escalar, $\gamma: I \to \mathbb{R}^1$).
            \item \textbf{Función $\vec{F}$ y dominio $U$:} La función que define la dinámica es $\vec{F}(t, z) = z^2 - z$, la cual está definida de forma continua sobre el conjunto abierto $U = \mathbb{R}^2$.
            \item \textbf{Función propuesta:} $\gamma(t) = \frac{1}{1 + Ce^t}$, donde $C$ es una constante real arbitraria.
        \end{itemize}
    \end{itemize}

    \subsubsection*{Verificación de las Condiciones}

    \begin{itemize}[leftmargin=*]
        \item \textbf{Regularidad $C^k$ ($C^1$):}\\
        Reescribimos la función propuesta usando exponentes negativos para facilitar su derivación mediante la regla de la cadena: $\gamma(t) = (1 + Ce^t)^{-1}$.\\
        Calculamos su primera derivada con respecto al tiempo:
        \begin{align*}
            \frac{d}{dt}\gamma(t) &= -1(1 + Ce^t)^{-2} \cdot \frac{d}{dt}(1 + Ce^t) \\
            \frac{d}{dt}\gamma(t) &= -1(1 + Ce^t)^{-2} \cdot (Ce^t) \\
            \frac{d}{dt}\gamma(t) &= \frac{-Ce^t}{(1 + Ce^t)^2}
        \end{align*}
        Mientras el denominador no se anule, esta función exponencial es infinitamente diferenciable ($C^\infty$), lo que garantiza que es de clase $C^1$.

        \item \textbf{Condición i) $I$ es un intervalo abierto:}\\
        El dominio temporal depende del valor de la constante $C$. Para evitar la división por cero, requerimos que $1 + Ce^t \neq 0$, es decir, $Ce^t \neq -1$. \\
        Si elegimos un $C \geq 0$, el denominador nunca será cero y el intervalo abierto es toda la recta real $I = (-\infty, \infty) = \mathbb{R}$. Si $C < 0$, el intervalo abierto se restringe a los valores donde $t \neq \ln(-1/C)$. En cualquiera de los casos, existe un intervalo abierto válido $I$.

        \item \textbf{Condición ii) Pertenece al dominio $U$ ($\vec{v}(t) \in U, \forall t \in I$):}\\
        Construimos el vector de estado para $k=1$:
        \[ \vec{v}(t) = (t, \gamma(t)) \]
        Dado que $U = \mathbb{R}^2$, el vector $\vec{v}(t)$ pertenece a $U$ para cualquier instante $t \in I$ en el que la función esté bien definida.

        \item \textbf{Condición iii) Satisfacción de la ecuación ($\frac{d}{dt}\gamma(t) = \vec{F}(\vec{v}(t))$):}
        \begin{itemize}[leftmargin=*]
            \item \textbf{Lado Izquierdo (derivada previamente calculada):}
            \[ \frac{d}{dt}\gamma(t) = \frac{-Ce^t}{(1 + Ce^t)^2} \]
            
            \item \textbf{Lado Derecho ($\vec{F}(\vec{v}(t))$):}\\
            Evaluamos la función en forma normal sustituyendo $\gamma(t)$:
            \begin{align*}
                \vec{F}(\vec{v}(t)) &= [\gamma(t)]^2 - \gamma(t) \\
                \vec{F}(\vec{v}(t)) &= \left( \frac{1}{1 + Ce^t} \right)^2 - \left( \frac{1}{1 + Ce^t} \right)
            \end{align*}
            Para restar estas fracciones, encontramos el denominador común $(1 + Ce^t)^2$:
            \[ \vec{F}(\vec{v}(t)) = \frac{1}{(1 + Ce^t)^2} - \frac{1 + Ce^t}{(1 + Ce^t)^2} \]
            Restamos los numeradores directamente:
            \[ \vec{F}(\vec{v}(t)) = \frac{1 - (1 + Ce^t)}{(1 + Ce^t)^2} = \frac{1 - 1 - Ce^t}{(1 + Ce^t)^2} = \frac{-Ce^t}{(1 + Ce^t)^2} \]
            
            \item \textbf{Identidad:}
            \[ \frac{-Ce^t}{(1 + Ce^t)^2} = \frac{-Ce^t}{(1 + Ce^t)^2} \]
        \end{itemize}
    \end{itemize}

    \vspace{0.3cm}
    \noindent Por lo tanto, hemos demostrado que la función $z(t) = \frac{1}{1+Ce^t}$ es efectivamente una solución de la Ecuación de Bernoulli.
